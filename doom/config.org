(defun jecs/sp-kill-sexp-region ()
  "Kill the text between sp-beginning-of-sexp and sp-end-of-sexp."
  (interactive)
  (save-excursion
    (let ((beg (progn (sp-beginning-of-sexp) (point)))
          (end (progn (sp-end-of-sexp) (point))))
      (kill-region beg end))))

(defun jecs/new-emacs-instance ()
  "Start a new independent Emacs.app instance (macOS)."
  (interactive)
  (cond
   ((eq system-type 'darwin)
    ;; Launch Emacs.app independently
    (start-process "emacs-new" nil "open" "-na" "Emacs"))
   (t
    (let ((exe (expand-file-name invocation-name invocation-directory)))
      (start-process-shell-command
       "emacs-new" nil (format "nohup %s >/dev/null 2>&1 &"
                               (shell-quote-argument exe)))))))
#+end_src

  (interactive)

*** jecs/delete-project-git-lock-file

#+begin_src emacs-lisp :results none :tangle yes
(defun jecs/delete-project-git-lock-file (&optional force)
  "Delete the current project's Git index.lock file.

With prefix arg FORCE, delete without confirmation."
  (interactive "P")
  (require 'subr-x)

  (let* ((root
          (or (when (fboundp 'projectile-project-root)
                (ignore-errors (projectile-project-root)))
              (when (fboundp 'project-current)
                (when-let ((project (project-current nil)))
                  (project-root project)))
              (vc-root-dir)))
         (lock-file
          (when root
            (let ((default-directory root))
              (expand-file-name
               (string-trim
                (shell-command-to-string
                 "git rev-parse --git-path index.lock 2>/dev/null"))
               root)))))

    (cond
     ((not root)
      (user-error "Not inside a project"))

     ((or (not lock-file) (string-empty-p lock-file))
      (user-error "Could not resolve Git lock file"))

     ((not (file-exists-p lock-file))
      (message "No Git lock file exists: %s" lock-file))

     ((or force
          (yes-or-no-p (format "Delete Git lock file?\n%s\n" lock-file)))
      (delete-file lock-file)
      (message "Deleted Git lock file: %s" lock-file)))))

(defun jecs/apply-frame-size-smart (&optional frame)
  (let* ((frame (or frame (selected-frame)))
         (workarea (frame-monitor-workarea frame))
         (wa-left   (nth 0 workarea))
         (wa-top    (nth 1 workarea))
         (wa-width  (nth 2 workarea))
         (wa-height (nth 3 workarea)))
    (set-frame-parameter frame 'fullscreen nil)
    (set-frame-size frame
                    jecs/default-frame-cols
                    jecs/default-frame-rows)

    (let ((frame-width  (frame-outer-width frame))
          (frame-height (frame-outer-height frame)))
      (if (or (> frame-width wa-width)
              (> frame-height wa-height))
          (set-frame-parameter frame 'fullscreen 'maximized)
        (set-frame-position
         frame
         (+ wa-left jecs/small-display-margin-left)
         (+ wa-top
            (- wa-height frame-height)
            (- jecs/small-display-margin-bottom)))))))

(defun jecs/dired-grep (needle)
  "Recursively search for NEEDLE from the current Dired directory.
Use grep over TRAMP, otherwise use ripgrep."
  (interactive
   (list
    (read-string
     (if (use-region-p)
         (format "Search for (default %s): "
                 (buffer-substring-no-properties
                  (region-beginning) (region-end)))
       "Search for: ")
     nil nil
     (when (use-region-p)
       (buffer-substring-no-properties
        (region-beginning) (region-end))))))
  (unless (derived-mode-p 'dired-mode)
    (user-error "Not in a Dired buffer"))
  (let* ((dir default-directory)
         (quoted (shell-quote-argument needle))
         (cmd (if (file-remote-p dir)
                  (format "find . -type f -exec grep -nH -- %s {} +"
                          quoted)
                (format "rg -n --hidden -g '!.git' %s ."
                        quoted))))
    (grep cmd)))

(map! :map dired-mode-map
      "/" #'jecs/dired-grep
      :localleader
      :desc "Search here" "s" #'jecs/dired-grep)

(after! dired
  (remove-hook 'dired-mode-hook #'dired-omit-mode))

(defun jecs/rotate-text (prefix)
  "Rotate text in region using `rotate-text' at each word boundary.
If there is no region then call `rotate-text' normally."
  (interactive "p")
  (let ((numeric-prefix (if (null prefix) 1
                          (prefix-numeric-value prefix))))
    (if (not (use-region-p))
        (rotate-text numeric-prefix)
      (let ((case-fold-search 0)
            (start (region-beginning))
            (end (region-end)))
        (save-excursion
          (while (and (goto-char end)
                      (setq end (re-search-backward "\\b\\w" start t)))
            (condition-case nil
                (rotate-text numeric-prefix)
              (error nil))))))))

(defun jecs/rotate-text-backward (prefix)
  "Same as `jecs/rotate-text' but backwards."
  (interactive "p")
  (let ((numeric-prefix (if (null prefix) 1
                          (prefix-numeric-value prefix))))
    (jecs/rotate-text (- numeric-prefix))))

** Emacs LSP Booster

[[https://github.com/blahgeek/emacs-lsp-booster][emacs-lsp-booster]] improves the performenace of lsp-mode by wrapping the lsp with an executable that translates json messages to elisp bytecode so Emacs doesn't have to.

If =lsp-use-plists= isn't non-nil then =export LSP_USE_PLISTS=true= and run =doom sync --rebuild=.

No longer using since added by doom lsp module.

#+begin_src emacs-lisp :tangle no
(defun lsp-booster--advice-json-parse (old-fn &rest args)
  "Try to parse bytecode instead of json."
  (or
   (when (equal (following-char) ?#)
     (let ((bytecode (read (current-buffer))))
       (when (byte-code-function-p bytecode)
         (funcall bytecode))))
   (apply old-fn args)))

(advice-add (if (progn (require 'json)
                       (fboundp 'json-parse-buffer))
                'json-parse-buffer
              'json-read)
            :around
            #'lsp-booster--advice-json-parse)

(defun lsp-booster--advice-final-command (old-fn cmd &optional test?)
  "Prepend emacs-lsp-booster command to lsp CMD."
  (let ((orig-result (funcall old-fn cmd test?)))
    (if (and (not test?)                             ;; for check lsp-server-present?
             (not (file-remote-p default-directory)) ;; see lsp-resolve-final-command, it would add extra shell wrapper
             lsp-use-plists
             (not (functionp 'json-rpc-connection))  ;; native json-rpc
             (executable-find "emacs-lsp-booster"))
        (progn
          (message "Using emacs-lsp-booster for %s!" orig-result)
          (cons "emacs-lsp-booster" orig-result))
      orig-result)))
(advice-add 'lsp-resolve-final-command :around #'lsp-booster--advice-final-command)

#+begin_src emacs-lisp :results none :tangle yes
;;; Paths

(defun jecs/cmake--project-root (project)
  (file-name-as-directory
   (expand-file-name (project-root project))))

(defun jecs/cmake--cache-directory (project)
  (expand-file-name
   ".cache/cmake-builds"
   (jecs/cmake--project-root project)))

(defun jecs/cmake--state-file (project)
  (expand-file-name
   "state.el"
   (jecs/cmake--cache-directory project)))

(defun jecs/cmake--active-build-directory (project)
  (expand-file-name
   "build"
   (jecs/cmake--project-root project)))

(defun jecs/cmake--parked-build-directory (project preset)
  (expand-file-name
   preset
   (jecs/cmake--cache-directory project)))


;;; Project detection

(defun jecs/cmake--preset-project-p (project)
  "Return non-nil when PROJECT is a CMake project using presets."
  (let ((root (jecs/cmake--project-root project)))
    (and
     (file-exists-p (expand-file-name "CMakeLists.txt" root))
     (or
      (file-exists-p (expand-file-name "CMakePresets.json" root))
      (file-exists-p (expand-file-name "CMakeUserPresets.json" root))))))


;;; State

(defun jecs/cmake--read-state (project)
  (when-let* ((file (jecs/cmake--state-file project))
              ((file-exists-p file)))
    (with-temp-buffer
      (insert-file-contents file)
      (let ((read-eval nil))
        (read (current-buffer))))))

(defun jecs/cmake--write-state (project preset)
  (let ((file (jecs/cmake--state-file project)))
    (make-directory (file-name-directory file) t)
    (with-temp-file file
      (prin1 (list :active-preset preset) (current-buffer))
      (insert "\n"))))

(defun jecs/cmake--active-preset (project)
  (plist-get
   (jecs/cmake--read-state project)
   :active-preset))


;;; Projection integration

(defun jecs/cmake--read-configure-preset (project)
  (let ((default-directory (jecs/cmake--project-root project)))
    (if-let* ((presets
               (projection-cmake--list-presets-for-build-type 'configure)))
        (projection-cmake--read-preset
         "Select active CMake preset: "
         presets)
      (user-error "No CMake configure presets found"))))

(defun jecs/cmake--set-projection-preset (project preset)
  "Set Projection's configure preset."
  (projection-cmake-set-preset
   project
   'configure
   preset))

(defun jecs/cmake--clear-projection-caches (project)
  (dolist (cache-var
           '(projection-cmake-code-model
             projection-cmake-ctest-targets
             projection-cmake-build-target
             projection-cmake-ctest-target))
    (projection-cache-clear-single
     project
     cache-var
     projection--project-cache)))

(defun jecs/cmake--set-active-preset (project preset)
  (dolist (build-type '(configure build test package workflow))
    (projection-cmake-set-preset project build-type nil))

  (jecs/cmake--set-projection-preset project preset)
  (jecs/cmake--clear-projection-caches project)
  (jecs/cmake--write-state project preset))


;;; Run command

(defvar jecs/cmake--run-arguments
  (make-hash-table :test #'equal))

(defvar jecs/cmake--run-argument-history nil)

(defun jecs/cmake--executable-artifacts ()
  (seq-filter
   (lambda (artifact)
     (eq (alist-get 'type artifact)
         'cmake-executable))
   (projection-cmake-list-artifacts)))

(defun jecs/cmake--read-executable-artifact ()
  (let* ((artifacts
          (jecs/cmake--executable-artifacts))
         (candidates
          (mapcar
           (lambda (artifact)
             (cons
              (alist-get 'name artifact)
              artifact))
           artifacts)))
    (pcase candidates
      ('()
       (user-error "No CMake executable artifacts found"))
      (`((,_ . ,artifact))
       artifact)
      (_
       (cdr
        (assoc
         (completing-read
          "Select CMake executable: "
          candidates
          nil
          t)
         candidates))))))

(defun jecs/cmake--read-run-arguments (project artifact)
  (let* ((key
          (cons
           (jecs/cmake--project-root project)
           (alist-get 'arg0 artifact)))
         (previous
          (gethash key jecs/cmake--run-arguments))
         (arguments
          (read-shell-command
           "Run arguments: "
           previous
           'jecs/cmake--run-argument-history)))
    (puthash
     key
     arguments
     jecs/cmake--run-arguments)
    arguments))

(defun jecs/cmake-run ()
  "Run a CMake executable artifact."
  (interactive)
  (let* ((project
          (or (project-current nil)
              (user-error "Not inside a project")))
         (default-directory
          (jecs/cmake--project-root project))
         (artifact
          (jecs/cmake--read-executable-artifact))
         (executable
          (alist-get 'arg0 artifact))
         (arguments
          (jecs/cmake--read-run-arguments
           project
           artifact)))
    (compile
     (string-join
      (delq
       nil
       (list
        (shell-quote-argument executable)
        (unless (string-empty-p arguments)
          arguments)))
      " "))))


;;; Initialization

(defun jecs/cmake--ensure-active-preset (project)
  (if-let* ((preset (jecs/cmake--active-preset project)))
      (progn
        (jecs/cmake--set-projection-preset project preset)
        preset)

    (when (file-exists-p
           (jecs/cmake--active-build-directory project))
      (user-error
       "CMake build/ exists without managed preset state; remove build/ once before initializing"))

    (let* ((preset-config
            (jecs/cmake--read-configure-preset project))
           (preset
            (alist-get 'name preset-config)))
      (jecs/cmake--set-active-preset project preset)
      preset)))

(defun jecs/cmake--initialize-project (project)
  "Initialize CMake task state when PROJECT uses CMake presets."
  (when (jecs/cmake--preset-project-p project)
    (require 'projection-type-cmake)
    (jecs/cmake--ensure-active-preset project)))

(add-hook 'jecs/project-task-init-hook
          #'jecs/cmake--initialize-project)


;;; Preset switching

(defun jecs/cmake--switch-build-tree (project preset)
  "Switch PROJECT to PRESET."
  (let* ((current-preset
          (jecs/cmake--active-preset project))
         (active-directory
          (jecs/cmake--active-build-directory project))
         (current-directory
          (jecs/cmake--parked-build-directory
           project
           current-preset))
         (new-directory
          (jecs/cmake--parked-build-directory
           project
           preset)))

    (if (equal current-preset preset)
        (message "CMake preset already active: %s" preset)

      (when (file-exists-p current-directory)
        (user-error
         "Cached CMake build already exists for %s"
         current-preset))

      (make-directory
       (jecs/cmake--cache-directory project)
       t)

      (when (file-exists-p active-directory)
        (rename-file
         active-directory
         current-directory))

      (when (file-exists-p new-directory)
        (rename-file
         new-directory
         active-directory))

      (jecs/cmake--set-active-preset project preset)

      (message
       (if (file-exists-p active-directory)
           "Switched CMake preset: %s -> %s"
         "Switched CMake preset: %s -> %s; configure required")
       current-preset
       preset))))

(defun jecs/cmake-switch-preset ()
  "Switch the current project's active CMake preset."
  (interactive)
  (let ((project
         (or (project-current nil)
             (user-error "Not inside a project"))))

    (unless (jecs/cmake--preset-project-p project)
      (user-error "Current project is not a CMake preset project"))

    (require 'projection-type-cmake)
    (jecs/cmake--ensure-active-preset project)

    (let* ((preset-config
            (jecs/cmake--read-configure-preset project))
           (preset
            (alist-get 'name preset-config)))
      (jecs/cmake--switch-build-tree
       project
       preset))))


;;; compile-multi integration

(defun jecs/cmake--compile-multi-targets ()
  (when-let* ((project (project-current nil))
              ((jecs/cmake--preset-project-p project)))
    `(("cmake:switch-preset"
       :command ,#'jecs/cmake-switch-preset
       :annotation ,(format
                     "active:%s"
                     (or
                      (jecs/cmake--active-preset project)
                      "none"))))))

        projection-cmake-build-directory "build")

  (oset projection-project-type-cmake

        #'jecs/cmake-run))

  (add-to-list

(defun jecs/clang--build-static-command ()
  (let* ((compiler
          (jecs/clang--compiler))
         (flags
          (jecs/clang--flags))
         (sources
          (jecs/clang--source-files))
         (objects
          (mapcar
           #'jecs/clang--object-file
           sources))
         (compile-commands
          (mapcar
           (lambda (source)
             (jecs/clang--shell-command
              `(,compiler
                ,@flags
                "-c"
                ,source
                "-o"
                ,(jecs/clang--object-file source))))
           sources)))
    (format
     "mkdir -p %s && %s && %s"
     (shell-quote-argument
      (jecs/clang--build-directory))
     (string-join compile-commands " && ")
     (jecs/clang--shell-command
      `("ar"
        "rcs"
        ,(jecs/clang--static-library-file)
        ,@objects)))))

(defun jecs/clang--build-shared-command ()

   "mkdir -p %s && %s"

    (jecs/clang--build-directory))

    `(,(jecs/clang--compiler)
      ,@(jecs/clang--flags)
      ,@(if (eq system-type 'darwin)
            '("-dynamiclib")
          '("-fPIC" "-shared"))
      ,@(jecs/clang--source-files)
      "-o"
      ,(jecs/clang--shared-library-file)))))

* Windows

Some configuration for when I have to use windows.

#+begin_src emacs-lisp :results none
(when (eq system-type 'windows-nt)
  (progn
    ;; use vs code font
    (set-frame-font "Cascadia Code 12" nil t)

    ;; vterm doesnt work on windows. use eshell instead in packages.el!
    (use-package window
      :custom
      (display-buffer-alist
       ;; bottom slot -1
       '(("\\*doom:eshell-popup:.*\\*"
          (display-buffer-in-side-window)
          (window-height . 0.25)
          (side . bottom)
          (slot . -1)
          (window-parameters . ((no-delete-other-windows . t)))))))))
#+end_src
